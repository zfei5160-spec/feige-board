// 优先加载最新的 data.json 数据 fetch('data.json?v=' + new Date().getTime()) .then(response => response.json()) .then(res => { if (res && res.data && res.data.items) { // 提取飞书全量数据并更新看板页面 const rawRecords = res.data.items.map(item => item.fields); renderDashboard(rawRecords); } }) .catch(err => console.log('使用本地默认数据', err));
